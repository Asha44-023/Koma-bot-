name: Keep Scheduler Alive
on:
  schedule:
    - cron: '*/15 * * * *'
  workflow_dispatch:

jobs:
  keepalive:
    runs-on: ubuntu-latest
    permissions:
      actions: write
      contents: read
    steps:
      - uses: actions/checkout@v4
      - name: Trigger Scan V700
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          echo "Waking up Scan V700..."
          gh workflow run "Scan V700" --ref main
          echo "Triggered"
